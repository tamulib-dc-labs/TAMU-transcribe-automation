# Transcription: Parakeet-TDT (words) + Sortformer (speaker turns).
# Both ship in nemo_toolkit[asr], so this is one model stack, not two.
#
# Needs Python >= 3.11.4 (or 3.12). NeMo unpacks its own .nemo model files with
# tarfile's filter= argument, which does not exist before then - GCCcore/12.3.0
# ships 3.11.3 and every model load fails. Run `ml spider Python` to pick one.

nemo_toolkit[asr]>=2.4
torch

# Audio decode and resample. PyAV reads mp4/mkv in-process, so no ffmpeg
# subprocess is spawned per file.
soundfile>=0.12
soxr>=0.3
av>=12.0
numpy>=1.24

# Optional preprocessing. The denoise pass is off by default -- both models are
# trained on noise-augmented audio and normalise their own input.
noisereduce

# Download and upload
smbprotocol
yt-dlp
pandas
tqdm
